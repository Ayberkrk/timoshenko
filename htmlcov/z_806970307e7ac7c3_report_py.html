11def _result_parts(result: Any) -> tuple[Any, Any, Any, str, str]: 

12 """Extract the stable structure/modal/health view shared by result types.""" 

13 if not all(hasattr(result, name) for name in ("structure", "modal", "health")): 

14 raise TypeError("result must be a Timoshenko monitoring, session, or project result") 

15 structure, modal, health = result.structure, result.modal, result.health 

16 if ( 

17 not hasattr(structure, "structure_id") 

18 or not hasattr(modal, "modes") 

19 or not hasattr(health, "mode_changes") 

20 ): 

21 raise TypeError("result does not expose the Timoshenko analysis result contract") 

22 title = str(getattr(result, "project_name", "") or structure.name or structure.structure_id) 

23 method = str(getattr(result, "method", "") or modal.method) 

24 return structure, modal, health, title, method

27def _number(value: float | None, digits: int = 3) -> str: 

28 if value is None: 

29 return "-" 

30 number = float(value) 

31 if not math.isfinite(number): 

32 return "-" 

33 return f"{number:.{digits}f}"

36def _chart_svg(modal: Any, health: Any) -> str: 

37 modes = list(modal.modes) 

38 changes = {int(item.observed_mode_number): item for item in health.mode_changes} 

39 width, height, margin = 780, max(230, 84 + 44 * len(modes)), 56 

40 usable_width = width - 2 * margin 

41 max_frequency = max( 

42 [float(item.frequency_hz) for item in modes] 

43 + [float(item.reference_frequency_hz) for item in changes.values()] 

44 + [1.0] 

45 ) 

46 scale = usable_width / max_frequency 

47 pieces = [ 

48 f'<svg class="frequency-chart" viewBox="0 0 {width} {height}" role="img" aria-labelledby="chart-title chart-desc">', 

49 '<title id="chart-title">Reference and observed modal frequencies</title>', 

50 '<desc id="chart-desc">Horizontal lines compare model reference frequencies with the frequencies identified from observations.</desc>', 

51 f'<line x1="{margin}" y1="26" x2="{width - margin}" y2="26" stroke="#94a3b8" stroke-width="1"/>', 

52 f'<text x="{margin}" y="18" class="axis-label">0 Hz</text>', 

53 f'<text x="{width - margin}" y="18" text-anchor="end" class="axis-label">{_number(max_frequency, 1)} Hz</text>', 

54 ] 

55 for index, mode in enumerate(modes): 

56 y = 62 + index * 44 

57 change = changes.get(index + 1) 

58 label = f"Mode {change.mode_number}" if change is not None else "Unpaired" 

59 pieces.append(f'<text x="4" y="{y + 4}" class="mode-label">{label}</text>') 

60 if change is not None: 

61 reference_x = margin + float(change.reference_frequency_hz) * scale 

62 pieces.append( 

63 f'<line x1="{margin}" y1="{y - 5}" x2="{reference_x:.2f}" y2="{y - 5}" stroke="#94a3b8" stroke-width="5" stroke-linecap="round"/>' 

64 ) 

65 pieces.append(f'<circle cx="{reference_x:.2f}" cy="{y - 5}" r="5" fill="#475569"/>') 

66 observed_x = margin + float(mode.frequency_hz) * scale 

67 pieces.append( 

68 f'<line x1="{margin}" y1="{y + 7}" x2="{observed_x:.2f}" y2="{y + 7}" stroke="#2563eb" stroke-width="5" stroke-linecap="round"/>' 

69 ) 

70 pieces.append(f'<circle cx="{observed_x:.2f}" cy="{y + 7}" r="5" fill="#1d4ed8"/>') 

71 pieces.append( 

72 f'<text x="{width - margin + 8}" y="{y - 2}" class="value-label">{_number(mode.frequency_hz)} Hz</text>' 

73 ) 

74 pieces.extend( 

75 [ 

76 f'<circle cx="{margin + 8}" cy="{height - 19}" r="5" fill="#475569"/><text x="{margin + 19}" y="{height - 15}" class="legend">Model reference</text>', 

77 f'<circle cx="{margin + 170}" cy="{height - 19}" r="5" fill="#1d4ed8"/><text x="{margin + 181}" y="{height - 15}" class="legend">Observed</text>', 

78 "</svg>", 

79 ] 

80 ) 

81 return "".join(pieces)

84def to_html(result: Any, *, title: str | None = None) -> str: 

85 """Render a self-contained accessible HTML report with a small SVG chart. 

86 

87 Supported inputs are ``MonitoringResult``, ``SessionReport``, and 

88 ``ProjectRunResult``. Project paths/hashes and session timestamps are 

89 included when the input provides them. 

90 """ 

91 structure, modal, health, default_title, method = _result_parts(result) 

92 display_title = escape(str(title or default_title)) 

93 modes = list(modal.modes) 

94 changes = {int(item.observed_mode_number): item for item in health.mode_changes} 

95 rows: list[str] = [] 

96 for index, mode in enumerate(modes, start=1): 

97 change = changes.get(index) 

98 reference = None if change is None else float(change.reference_frequency_hz) 

99 shift = None if change is None else float(change.change_pct) 

100 damping = getattr(mode, "damping_ratio", None) 

101 rows.append( 

102 "<tr>" 

103 f'<th scope="row">{change.mode_number if change is not None else "unpaired"}</th>' 

104 f"<td>{_number(reference)}</td>" 

105 f"<td>{_number(float(mode.frequency_hz))}</td>" 

106 f"<td>{_number(shift, 2)}{'%' if shift is not None else ''}</td>" 

107 f"<td>{_number(damping, 4)}</td>" 

108 "</tr>" 

109 ) 

110 if not rows: 

111 rows.append('<tr><td colspan="5">No usable modal frequencies were identified.</td></tr>') 

112 

113 metadata: list[str] = [ 

114 f"<dt>Structure</dt><dd>{escape(str(structure.structure_id))}</dd>", 

115 f"<dt>Method</dt><dd>{escape(method)}</dd>", 

116 f"<dt>Analysis status</dt><dd>{escape(str(modal.status))}</dd>", 

117 f"<dt>Health evidence</dt><dd>{escape(str(health.status))}</dd>", 

118 ] 

119 if hasattr(result, "event_time_s"): 

120 metadata.append( 

121 f"<dt>Window end (Unix time)</dt><dd>{_number(result.event_time_s, 3)}</dd>" 

122 ) 

123 if hasattr(result, "source_sha256"): 

124 metadata.append( 

125 f"<dt>Input SHA-256</dt><dd><code>{escape(str(result.source_sha256))}</code></dd>" 

126 ) 

127 metadata.append( 

128 f"<dt>Manifest SHA-256</dt><dd><code>{escape(str(result.manifest_sha256))}</code></dd>" 

129 ) 

130 limitations = "".join(f"<li>{escape(str(item))}</li>" for item in health.limitations) 

131 notes = "".join(f"<li>{escape(str(item))}</li>" for item in getattr(modal, "notes", ())) 

132 summary = escape(str(health.evidence_summary)) 

133 threshold = getattr(health, "review_threshold_pct", None) 

134 if health.review_recommended: 

135 review = f"Review recommended (frequency drop of at least {_number(threshold, 2)}%)" 

136 elif threshold is None: 

137 review = "No review threshold supplied; no review flag evaluated" 

138 else: 

139 review = f"No paired mode dropped by {_number(threshold, 2)}% or more" 

140 chart = _chart_svg(modal, health) 

141 return f"""<!doctype html> 

142<html lang="en"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1"> 

143<title>{display_title} - Timoshenko report</title> 

144<style> 

145:root{{color-scheme:light;--ink:#172033;--muted:#536178;--line:#dbe2ea;--blue:#1d4ed8;--paper:#fff;--wash:#f4f7fb}} 

146*{{box-sizing:border-box}}body{{margin:0;background:var(--wash);color:var(--ink);font:16px/1.5 system-ui,-apple-system,"Segoe UI",sans-serif}} 

147main{{max-width:980px;margin:36px auto;padding:36px;background:var(--paper);border:1px solid var(--line);border-radius:14px}} 

148h1{{margin:0 0 8px;font-size:2rem}}h2{{margin:30px 0 10px;font-size:1.2rem}}.subtle,.axis-label,.legend{{color:var(--muted)}} 

149dl{{display:grid;grid-template-columns:minmax(150px,220px) 1fr;gap:8px 16px;margin:20px 0}}dt{{font-weight:650}}dd{{margin:0;overflow-wrap:anywhere}} 

150.summary{{padding:14px 16px;border-left:4px solid var(--blue);background:#eff6ff}}table{{width:100%;border-collapse:collapse}}th,td{{text-align:left;padding:10px 12px;border-bottom:1px solid var(--line);font-variant-numeric:tabular-nums}}thead{{background:var(--wash)}} 

151.frequency-chart{{width:100%;height:auto;overflow:visible}}.mode-label,.value-label{{font:12px system-ui,sans-serif;fill:var(--ink)}}.axis-label,.legend{{font:11px system-ui,sans-serif;fill:var(--muted)}}code{{font-size:.85em;overflow-wrap:anywhere}} 

152@media(max-width:640px){{main{{margin:0;padding:20px;border-radius:0}}dl{{grid-template-columns:1fr;gap:2px}}dd{{margin-bottom:10px}}}} 

153</style></head><body><main> 

154<p class="subtle">Timoshenko Engine · analysis report</p><h1>{display_title}</h1> 

155<p class="summary">{summary}<br><strong>{escape(review)}</strong></p> 

156<dl>{"".join(metadata)}</dl> 

157<h2>Modal frequency comparison</h2>{chart} 

158<table><thead><tr><th>Mode</th><th>Reference (Hz)</th><th>Observed (Hz)</th><th>Difference</th><th>Damping ratio</th></tr></thead><tbody>{"".join(rows)}</tbody></table> 

159<h2>Method notes</h2><ul>{notes or "<li>No additional method notes.</li>"}</ul> 

160<h2>Interpretation limits</h2><ul>{limitations or "<li>No limitation notes were supplied.</li>"}</ul> 

161</main></body></html>"""

164def save_html(result: Any, path: str | Path, *, title: str | None = None) -> Path: 

165 """Write a UTF-8 self-contained HTML report and return its resolved path.""" 

166 destination = Path(path).expanduser().resolve() 

167 destination.parent.mkdir(parents=True, exist_ok=True) 

168 destination.write_text(to_html(result, title=title), encoding="utf-8") 

169 return destination
